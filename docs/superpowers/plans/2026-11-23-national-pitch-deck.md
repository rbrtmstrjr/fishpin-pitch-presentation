```python
"""Bring brand assets and pinned libraries into the FishPin deck.

def fetch(url: str) -> bytes:
    req = urllib.request.Request(url, headers={"User-Agent": "deck-asset-import"})
    with urllib.request.urlopen(req, timeout=60) as resp:
        if resp.status != 200:
            raise RuntimeError(f"{url} returned HTTP {resp.status}")
        return resp.read()


def out(rel: str) -> Path:
    path = ASSETS / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    return path


def source_ready(src: Path, dst: Path) -> bool:
    """True if src exists. If src is gone but dst was imported before, keep dst."""
    if src.exists():
        return True
    if dst.exists():
        print(f"keep  {dst.relative_to(ROOT)} (source gone: {src})")
        return False
    sys.exit(f"missing source: {src}")


def main() -> None:
    for rel, url in VENDOR.items():
        out(rel).write_bytes(fetch(url))
        print(f"fetch {rel}")

    icons = {}
    for name in ICONS:
        svg = fetch(ICON_URL.format(name)).decode("utf-8")
        svg = re.sub(r"<!--.*?-->", "", svg, flags=re.S)
        svg = re.sub(r"\s+", " ", svg).strip().replace("> <", "><")
        svg = re.sub(r'\s(?:width|height)="24"', "", svg)
        icons[name] = svg
    out("icons.js").write_text(
        "/* Lucide icons (lucide-static 0.462.0, ISC licence). Generated by scripts/import_assets.py */\n"
        "window.ICONS = " + json.dumps(icons, indent=0) + ";\n",
        encoding="utf-8",
    )
    print(f"write icons.js ({len(icons)} icons)")

    for rel, src in COPY.items():
        dst = out(rel)
        if source_ready(src, dst):
            shutil.copyfile(src, dst)
            print(f"copy  {rel}")

    svg = CONTOUR_SRC.read_text(encoding="utf-8")
    svg = re.sub(r'stroke="[^"]+"', f'stroke="{CONTOUR_STROKE}"', svg)
    svg = re.sub(r"stroke:\s*[^;\"]+", f"stroke:{CONTOUR_STROKE}", svg)
    if CONTOUR_STROKE not in svg:
        svg = svg.replace("<svg ", f'<svg stroke="{CONTOUR_STROKE}" fill="none" ', 1)
    out(CONTOUR_DST).write_text(svg, encoding="utf-8")
    print(f"write {CONTOUR_DST}")

    hero = Image.open(HERO_SRC)
    hero = hero.resize((HERO_WIDTH, round(hero.height * HERO_WIDTH / hero.width)), Image.LANCZOS)
    hero.save(out(HERO_DST), optimize=True)
    print(f"write {HERO_DST}")

    for rel, shot in SCREENS.items():
        img = Image.open(SHOTS / shot).convert("RGB")
        img = img.resize((SCREEN_WIDTH, round(img.height * SCREEN_WIDTH / img.width)), Image.LANCZOS)
        img.save(out(rel), optimize=True)
        print(f"write {rel}")

    for rel, src in PHOTOS.items():
        dst = out(rel)
        if not source_ready(src, dst):
            continue
        img = ImageOps.exif_transpose(Image.open(src)).convert("RGB")
        img.thumbnail((PHOTO_MAX, PHOTO_MAX), Image.LANCZOS)
        img.save(dst, quality=82, optimize=True, progressive=True)
        print(f"write {rel}")


if __name__ == "__main__":
    main()
```

```python
"""Claims guard for the FishPin national pitch deck (spec section 7).

Usage:  python scripts/check_claims.py FILE [FILE ...]
  *.html  checks the visible text (styles, scripts and comments removed, speaker
          notes kept) plus <img> sources (no competitor logos). Line numbers refer
          to the extracted text.
  other   checks the file as plain text, e.g. tools/qa/out/rendered.txt written by
          the browser QA after every number has been filled in.
Prints one line per problem (file:line: rule: text) and exits 1 if there is any.
"""
from __future__ import annotations

import html
import re
import sys
from dataclasses import dataclass
from pathlib import Path

I = re.IGNORECASE


@dataclass(frozen=True)
class Rule:
    name: str
    pattern: re.Pattern
    allow: re.Pattern | None = None     # the line is fine if it also matches this
    require: re.Pattern | None = None   # a hit only counts if the line also matches this
    strip: re.Pattern | None = None     # removed from the line before matching
    group_ok: str | None = None         # a hit is fine when group(1) equals this


ORGS = re.compile(
    r"BFAR|NAMRIA|\bLGUs?\b|Coast Guard|cooperative|\bco-ops?\b|Municipal Agriculture|Department of Agriculture", I)

RULES = [
    Rule("em-dash", re.compile("—")),
    Rule("sector-size", re.compile(r"(?<![\d.])(\d+(?:\.\d+)?)\s*(?:million|M)\b[^.\n]{0,40}fisher", I), group_ok="1.1"),
    Rule("seventy-percent-unlabelled", re.compile(r"\b70\s*(?:%|percent|per cent)", I), allow=re.compile(r"assum", I)),
    Rule("university", re.compile(r"universit|college|campus|\bMinSU\b|Mindoro State", I)),
    Rule("free", re.compile(r"\bfree\b|\blibre\b|walang bayad", I),
         strip=re.compile(r"(?:\d*-?day )?free trial|free to try", I)),
    Rule("offline-absolute", re.compile(r"100\s*%\s*offline|fully offline|completely offline|entirely offline", I)),
    Rule("old-price", re.compile("(?:PHP|Php|₱)\\s*499(?![\\d,])")),
    Rule("partner-claim", re.compile(r"partner|endorse|katuwang|accredit|backed by|supported by|in cooperation with", I),
         require=ORGS),
    Rule("tagalog-app", re.compile(r"(?:Tagalog|Filipino)(?:-language)?\s+(?:app|interface|UI|screens)", I),
         allow=re.compile(r"plan|next|translation", I)),
    Rule("ios-shipped", re.compile(r"\biOS\b|App Store|iPhone", I),
         allow=re.compile(r"plan|December 2026|Dec 2026|next|release|build", I)),
    Rule("paywall-promise", re.compile(r"behind a paywall", I)),
    Rule("gemini", re.compile(r"Gemini(?! API)")),
]
LOGO = re.compile(r"<img\b[^>]*\bsrc=[\"'][^\"']*(?:navionics|fishbrain|garmin|c-?map)", I)
BLOCK_TAGS = re.compile(
    r"</?(?:p|div|li|ul|ol|dl|dt|dd|h[1-6]|section|aside|tr|td|th|table|thead|tbody|figure|figcaption|header|footer|br)\b[^>]*>", I)


def html_to_text(src: str) -> str:
    src = re.sub(r"(?is)<(style|script)\b.*?</\1>", " ", src)
    src = re.sub(r"(?s)<!--.*?-->", " ", src)
    src = BLOCK_TAGS.sub("\n", src)
    src = re.sub(r"<[^>]+>", " ", src)
    text = html.unescape(src)
    return "\n".join(re.sub(r"[ \t]+", " ", line).strip() for line in text.splitlines())


def check_text(text: str) -> list[tuple[int, str, str]]:
    hits = []
    for number, line in enumerate(text.splitlines(), 1):
        for rule in RULES:
            subject = rule.strip.sub("", line) if rule.strip else line
            for match in rule.pattern.finditer(subject):
                if rule.group_ok is not None and match.group(1) == rule.group_ok:
                    continue
                if rule.allow is not None and rule.allow.search(line):
                    continue
                if rule.require is not None and not rule.require.search(line):
                    continue
                hits.append((number, rule.name, line.strip()[:160]))
                break
    return hits


def check_html(src: str) -> list[tuple[int, str, str]]:
    hits = check_text(html_to_text(src))
    for match in LOGO.finditer(src):
        hits.append((src.count("\n", 0, match.start()) + 1, "competitor-logo", match.group(0)[:160]))
    return hits


def main(argv: list[str]) -> int:
    if not argv:
        print(__doc__)
        return 2
    total = 0
    for name in argv:
        path = Path(name)
        src = path.read_text(encoding="utf-8")
        hits = check_html(src) if path.suffix.lower() == ".html" else check_text(src)
        for line, rule, text in hits:
            print(f"{name}:{line}: {rule}: {text}")
        total += len(hits)
    print(f"claims check: {total} problem(s)")
    return 1 if total else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
```

```js
/* Slide facts that must hold once the slide has settled. Each returns '' or an error. */
const CHECKS = {
  problem: () => {
    const dots = document.querySelectorAll('#problem .dot').length;
    const lit = document.querySelectorAll('#problem .dot.gps').length;
    if (dots !== DECK.market.barangayFishermen) return `expected ${DECK.market.barangayFishermen} dots, found ${dots}`;

  novelty: () => document.querySelectorAll('#novelty .track span').length >= DECK.fishNames.length * 2 ? '' : 'fish-name rows are not filled',
  traction: () => document.querySelectorAll('#traction .ms').length === DECK.traction.milestones.length ? '' : 'timeline does not match DECK.traction.milestones',
  model: () => document.querySelectorAll('#model .bar-row').length === DECK.competitors.length + 1 ? '' : 'cost bars do not match DECK.competitors plus FishPin',
  market: () => document.querySelectorAll('#market .ring').length === 3 ? '' : 'market needs 3 rings',

};

const failures = [];
const warnings = [];
const fail = (where, msg) => failures.push(`${where}: ${msg}`);
const warn = (where, msg) => warnings.push(`${where}: ${msg}`);
const pad = n => String(n).padStart(2, '0');
const sleep = ms => new Promise(r => setTimeout(r, ms));
const run = id => !only || only.includes(id);

const browser = await chromium.launch({ channel: 'chrome' });

async function open(query = '', viewport = { width: 1920, height: 1080 }) {
  const context = await browser.newContext({ viewport });
  const errors = [];
  await context.route('**/*', route => {
    const url = route.request().url();
    if (/^(file|data|blob|about):/.test(url)) return route.continue();
    errors.push(`network request blocked: ${url}`);
    return route.abort();
  });
  const page = await context.newPage();
  page.on('pageerror', e => errors.push(`page error: ${e.message}`));
  page.on('console', m => { if (m.type() === 'error') errors.push(`console error: ${m.text()}`); });
  page.on('requestfailed', r => { if (!/^https?:/.test(r.url())) errors.push(`failed to load: ${r.url()}`); });
  let ready = true;
  try {
    await page.goto(DECK_URL + query);
    await page.waitForFunction(() => window.__deckReady === true, null, { timeout: 15000 });
  } catch (e) {
    ready = false;
    errors.push(`deck did not become ready: ${e.message.split('\n')[0]}`);
  }
  return { context, page, errors, ready };
}

const slideList = page => page.evaluate(() => [...document.querySelectorAll('.reveal .slides > section')]
  .map((s, i) => ({ i, id: s.id, appendix: s.hasAttribute('data-appendix'), time: Number(s.dataset.time || 0) })));
const goTo = (page, id) => page.evaluate(id => {
  const all = [...document.querySelectorAll('.reveal .slides > section')];
  Reveal.slide(all.findIndex(s => s.id === id));
}, id);
const currentId = page => page.evaluate(() => Reveal.getCurrentSlide().id);

/* Runs inside the page: measures the current slide once it has settled. */
function inspect(minFont) {
  const sec = Reveal.getCurrentSlide();
  const main = !sec.hasAttribute('data-appendix');
  const box = sec.getBoundingClientRect();
  const k = box.width / 1920;
  const name = el => el.id ? `#${el.id}` : `${el.tagName.toLowerCase()}${el.classList.length ? '.' + [...el.classList].join('.') : ''}`;
  const shown = el => { const cs = getComputedStyle(el); return cs.display !== 'none' && cs.visibility !== 'hidden'; };
  const exempt = el => el.closest('aside.notes, [data-bleed], [data-ui]');
  const els = [...sec.querySelectorAll('*')].filter(el => !el.closest('aside.notes'));
  const overflow = els.filter(el => {
    if (exempt(el) || !shown(el)) return false;
    const b = el.getBoundingClientRect();
    if (!b.width || !b.height) return false;
    const l = (b.left - box.left) / k, t = (b.top - box.top) / k, r = (b.right - box.left) / k, btm = (b.bottom - box.top) / k;
    return l < -2 || t < -2 || r > 1922 || btm > 1082;
  }).slice(0, 5).map(name);
  const tiny = main ? els.filter(el => {
    if (exempt(el) || !shown(el)) return false;
    const ownText = [...el.childNodes].some(n => n.nodeType === 3 && n.textContent.trim());
    return ownText && parseFloat(getComputedStyle(el).fontSize) < minFont - 0.5;
  }).slice(0, 5).map(el => `${name(el)} (${getComputedStyle(el).fontSize})`) : [];
  const unbound = [...sec.querySelectorAll('[data-k]')].filter(el => !el.textContent.trim() || /undefined|NaN/.test(el.textContent)).map(el => el.dataset.k);
  const broken = [...sec.querySelectorAll('img')].filter(im => !im.complete || !im.naturalWidth).map(im => im.getAttribute('src'));
  const hidden = [...sec.querySelectorAll('[data-rise]')].filter(el => parseFloat(getComputedStyle(el).opacity) < 0.95).map(name);
  const counter = document.getElementById('slideCounter')?.textContent.trim() ?? '';
  const notes = sec.querySelector('aside.notes')?.textContent.trim() ?? '';
  return { overflow, tiny, unbound, broken, hidden, counter, notes, text: sec.innerText.trim() };
}

async function checkSlides() {
  const { context, page, errors, ready } = await open();
  if (!ready) { errors.forEach(e => fail('runtime', e)); await context.close(); return []; }
  const slides = await slideList(page);
  const mains = slides.filter(s => !s.appendix);
  if (only) only.filter(id => !slides.some(s => s.id === id)).forEach(id => fail('slides', `no slide with id "${id}"`));

  const data = await page.evaluate(() => ({ problems: DECK.verify(), pending: DECK.pending() }));
  data.problems.forEach(p => fail('DECK.verify', p));
  if (data.pending.length) warn('DECK.pending', `not yet confirmed: ${data.pending.join(', ')}`);

  rmSync(OUT, { recursive: true, force: true });
  mkdirSync(OUT, { recursive: true });
  let rendered = '';
  for (const s of slides) {
    if (!run(s.id)) continue;
    await goTo(page, s.id);
    const wait = WAIT[s.id] ?? DEFAULT_WAIT;
    await sleep(wait);
    const r = await page.evaluate(inspect, MIN_FONT);
    const where = `${pad(s.i + 1)} ${s.id}`;
    if (r.overflow.length) fail(where, `outside the 1920x1080 frame: ${r.overflow.join(', ')}`);
    if (r.tiny.length) fail(where, `text below ${MIN_FONT}px: ${r.tiny.join(', ')}`);
    if (r.unbound.length) fail(where, `unfilled data: ${r.unbound.join(', ')}`);
    if (r.broken.length) fail(where, `broken images: ${r.broken.join(', ')}`);
    if (r.hidden.length) fail(where, `still hidden after ${wait}ms: ${r.hidden.join(', ')}`);
    if (s.appendix) {
      if (!r.counter.startsWith('Q&A')) fail(where, `counter shows "${r.counter}", expected Q&A`);
    } else {
      const want = `${pad(mains.findIndex(m => m.id === s.id) + 1)} / ${pad(mains.length)}`;
      if (r.counter !== want) fail(where, `counter shows "${r.counter}", expected "${want}"`);
      if (!s.time) fail(where, 'main slide has no data-time');
      const words = r.notes ? r.notes.split(/\s+/).length : 0;
      const budget = Math.ceil(s.time * WORDS_PER_MINUTE / 60);
      if (words > budget) fail(where, `speaker notes are ${words} words; budget is ${budget} for ${s.time}s`);
      if (has('notes') && !words) fail(where, 'speaker notes missing');
    }
    if (CHECKS[s.id]) {
      const msg = await page.evaluate(`(${CHECKS[s.id]})()`);
      if (msg) fail(where, msg);
    }
    await page.screenshot({ path: join(OUT, `${pad(s.i + 1)}-${s.id}.png`) });
    rendered += `=== ${s.id} [${s.appendix ? 'appendix' : 'main'}] ===\n${r.text}\n--- notes ---\n${r.notes}\n\n`;
  }
  if (!only) {
    const total = mains.reduce((t, s) => t + s.time, 0);
    if (total > MAIN_SECONDS) fail('timing', `main slides add up to ${total}s; budget is ${MAIN_SECONDS}s`);
  }
  writeFileSync(join(OUT, 'rendered.txt'), rendered, 'utf8');
  await checkInteractions(page, slides);
  errors.forEach(e => fail('runtime', e));
  await context.close();
  return slides;
}

async function checkInteractions(page, slides) {
  const ids = new Set(slides.map(s => s.id));
  const attr = (sel, a) => page.evaluate(([sel, a]) => document.querySelector(sel)?.getAttribute(a), [sel, a]);

  if (ids.has('offline') && run('offline')) {
    await goTo(page, 'offline'); await sleep(3000);
    if (await attr('#airSwitch', 'aria-checked') !== 'true') fail('offline', 'airplane switch did not flip on by itself');
    await page.keyboard.press('m'); await sleep(500);
    if (await attr('#airSwitch', 'aria-checked') !== 'false') fail('offline', 'M key did not toggle airplane mode');
    await page.click('#airSwitch'); await sleep(500);
    if (await attr('#airSwitch', 'aria-checked') !== 'true') fail('offline', 'click did not toggle airplane mode');
  }

  if (ids.has('impact') && run('impact')) {
    await goTo(page, 'impact'); await sleep(5200);
    if (await attr('#sos', 'data-state') !== 'sent') fail('impact', 'SOS demo did not finish on its own');
    const b = await page.locator('#sosBtn').boundingBox();
    await page.mouse.move(b.x + b.width / 2, b.y + b.height / 2);
    await page.mouse.down(); await sleep(800); await page.mouse.up(); await sleep(700);
    if (await attr('#sos', 'data-state') !== 'idle') fail('impact', 'a short press did not reset the SOS ring');
    await page.mouse.down(); await sleep(3400); await page.mouse.up(); await sleep(400);
    if (await attr('#sos', 'data-state') !== 'sent') fail('impact', 'a full hold did not send');
  }

  const appx = slides.find(s => s.appendix);
  const home = slides.find(s => s.id === 'financials') ?? slides.find(s => !s.appendix && s.id !== 'close');
  if (appx && home && !only) {
    await goTo(page, home.id); await sleep(500);
    await page.keyboard.press('q'); await sleep(300);
    if (!(await page.evaluate(() => document.getElementById('qmenu').classList.contains('open')))) fail('q-menu', 'Q did not open the menu');
    await page.click(`#qmenu [data-target="${appx.id}"]`); await sleep(700);
    if (await currentId(page) !== appx.id) fail('q-menu', `menu click did not jump to ${appx.id}`);
    await page.keyboard.press('Escape'); await sleep(700);
    if (await currentId(page) !== home.id) fail('q-menu', `Esc did not return to ${home.id}`);
    await page.keyboard.press('q'); await sleep(300); await page.keyboard.press('1'); await sleep(700);
    if (await currentId(page) !== appx.id) fail('q-menu', 'number key 1 did not open the first backup slide');
    await page.keyboard.press('Backspace'); await sleep(700);
    if (await currentId(page) !== home.id) fail('q-menu', `Backspace did not return to ${home.id}`);
  }

  if (ids.has('close') && !only) {
    await goTo(page, 'close'); await sleep(500);
    for (const key of ['ArrowRight', 'PageDown', 'Space']) {
      await page.keyboard.press(key); await sleep(400);
      if (await currentId(page) !== 'close') fail('clicker', `${key} on the last main slide left it`);
    }
  }

  if (!only) {
    await page.keyboard.press('t'); await sleep(200);
    if (await page.evaluate(() => document.getElementById('timer').classList.contains('hidden'))) fail('timer', 'T did not show the timer');
    await sleep(1300); await page.keyboard.press('r'); await sleep(300);
    const label = await page.evaluate(() => document.getElementById('timerLabel').textContent);
    if (label !== '0:00') fail('timer', `R did not reset the timer (shows ${label})`);
    await page.keyboard.press('t');
  }
}

async function checkStatic() {
  const { context, page, errors, ready } = await open('?static=1');
  if (ready) {
    for (const s of await slideList(page)) {
      if (!run(s.id)) continue;
      await goTo(page, s.id); await sleep(250);
      const r = await page.evaluate(inspect, MIN_FONT);
      if (r.hidden.length) fail(`static ${s.id}`, `hidden in static mode: ${r.hidden.join(', ')}`);
      if (r.unbound.length) fail(`static ${s.id}`, `unfilled data: ${r.unbound.join(', ')}`);
    }
  }
  errors.forEach(e => fail('static runtime', e));
  await context.close();
}

async function checkViewports() {
  for (const vp of [{ width: 1280, height: 800 }, { width: 1024, height: 768 }]) {
    const { context, page, errors } = await open('', vp);
    const slides = await slideList(page);
    for (const s of [slides.find(x => !x.appendix), slides.find(x => x.appendix)].filter(Boolean)) {
      await goTo(page, s.id); await sleep(1200);
      const bg = await page.evaluate(() => getComputedStyle(document.documentElement).backgroundColor);
      const want = s.appendix ? 'rgb(238, 244, 251)' : 'rgb(6, 18, 51)';
      if (bg !== want) fail(`viewport ${vp.width}x${vp.height}`, `letterbox on ${s.id} is ${bg}, expected ${want}`);
      await page.screenshot({ path: join(OUT, `vp-${vp.width}x${vp.height}-${s.id}.png`) });
    }
    errors.forEach(e => fail(`viewport ${vp.width}x${vp.height}`, e));
    await context.close();
  }
}

async function checkSpeakerView() {
  const { context, page, errors } = await open();
  try {
    const [popup] = await Promise.all([context.waitForEvent('page', { timeout: 6000 }), page.keyboard.press('s')]);
    await sleep(3000);
    const text = await popup.evaluate(() => document.body.innerText);
    if (!/notes/i.test(text)) warn('speaker view', 'opened but shows no notes panel');
    await popup.close();
  } catch (e) {
    warn('speaker view', `did not open over file:// (${e.message.split('\n')[0]}). Use the Vercel link or "python -m http.server" for speaker view.`);
  }
  errors.forEach(e => fail('speaker view runtime', e));
  await context.close();
}

async function exportBackups(slides) {
  rmSync(EXPORTS, { recursive: true, force: true });
  mkdirSync(join(EXPORTS, 'png'), { recursive: true });
  const still = await open('?static=1');
  for (const s of slides) {
    await goTo(still.page, s.id); await sleep(300);
    await still.page.screenshot({ path: join(EXPORTS, 'png', `${pad(s.i + 1)}-${s.id}.png`) });
  }
  still.errors.forEach(e => fail('export png', e));
  await still.context.close();

  const print = await open('?print-pdf');
  await print.page.waitForFunction(() => window.__pdfReady === true, null, { timeout: 30000 })
    .catch(() => fail('export pdf', 'Reveal never reported pdf-ready'));
  await sleep(1500);
  const file = join(EXPORTS, 'FishPin-Pitch.pdf');
  await print.page.pdf({ path: file, printBackground: true, preferCSSPageSize: true });
  const pages = (readFileSync(file, 'latin1').match(/\/Type\s*\/Page[^s]/g) || []).length;
  if (pages !== slides.length) fail('export pdf', `PDF has ${pages} pages, expected ${slides.length}`);
  print.errors.forEach(e => fail('export pdf', e));
  await print.context.close();
}

let slides = [];
try {
  slides = await checkSlides();
  if (slides.length) {
    await checkStatic();
    if (has('viewports')) await checkViewports();
    if (has('notes')) await checkSpeakerView();
    if (has('export')) await exportBackups(slides);
  }
} finally {
  await browser.close();
}
warnings.forEach(w => console.log(`warn  ${w}`));
failures.forEach(f => console.log(`FAIL  ${f}`));
console.log(`\n${slides.length} slide(s) checked, ${failures.length} failure(s), ${warnings.length} warning(s). Screenshots: tools/qa/out/`);
process.exit(failures.length ? 1 : 0);
```

```js
/* FishPin national pitch: presentation engine.
   Scripts load in this order: reveal.js, notes.js, gsap, assets/icons.js, deck-data.js, deck.js.
   On load: slide code builds data-driven markup (RENDER) -> icons are inlined -> every
   [data-k] element is filled from DECK -> Reveal starts -> each slide's entrance and
   signature motion (ANIM) run when it is shown.
   CSS always holds a slide's FINAL state. Animations only play into it, so static,
   print and reduced-motion modes need no special cases. */
(function () {
  'use strict';

  /* ---------- modes ---------- */
  const query = new URLSearchParams(location.search);
  const isPrint = /print-pdf/i.test(location.search);
  const isStatic = query.get('static') === '1';
  const isCheck = query.get('check') === '1';
  const still = isPrint || isStatic || window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  if (still) document.documentElement.classList.add('still');

  /* Headless print jobs can starve requestAnimationFrame, and Reveal's PDF layout waits on it.
     In print mode, race each frame against a short timeout so the layout always finishes. */
  if (isPrint) {
    const nativeRAF = window.requestAnimationFrame.bind(window);
    window.requestAnimationFrame = cb => {
      let done = false;
      const runOnce = ts => { if (!done) { done = true; cb(ts); } };
      const t = setTimeout(() => runOnce(performance.now()), 32);
      nativeRAF(runOnce);
      return t;
    };
  }

  /* ---------- helpers ---------- */
  const $ = (sel, root = document) => root.querySelector(sel);
  const $$ = (sel, root = document) => Array.from(root.querySelectorAll(sel));
  const pad = n => String(n).padStart(2, '0');
  const sections = () => $$('.reveal .slides > section');
  const mainSlides = () => sections().filter(s => !s.hasAttribute('data-appendix'));
  const nf = (v, dp = 0) => Number(v).toLocaleString('en-US', { minimumFractionDigits: dp, maximumFractionDigits: dp });

  /* ---------- number formats for [data-f] ---------- */
  const FORMAT = {
    text: v => String(v),
    int: v => nf(v),
    dec1: v => nf(v, 1),
    php: v => 'PHP ' + nf(v),
    php2: v => 'PHP ' + nf(v, 2),
    phpM: v => 'PHP ' + nf(v) + 'M',
    pct: v => nf(v, 2) + '%',
    pct1: v => nf(v, 1) + '%',
    share: v => nf(v * 100) + '%',
    yrs: v => nf(v, 2) + ' yrs',
    million: v => nf(v / 1e6, 1) + ' million',
    date: v => new Date(v + 'T00:00:00').toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' }),
  };
  const WHOLE = new Set(['int', 'php', 'phpM']);
  const format = (el, v) => (FORMAT[el.dataset.f] || FORMAT.text)(v);

  function bind(root) {
    $$('[data-k]', root).forEach(el => {
      const v = DECK.get(el.dataset.k);
      if (v === undefined || v === null || v === '') { console.error(`deck: no data for "${el.dataset.k}"`); return; }
      el.textContent = format(el, v);
      if (typeof v === 'number') el.dataset.v = String(v);
    });
  }

  function inlineIcons(root) {
    $$('[data-icon]', root).forEach(el => {
      const svg = window.ICONS && window.ICONS[el.dataset.icon];
      if (!svg) { console.error(`deck: missing icon "${el.dataset.icon}"`); return; }
      el.innerHTML = svg;
    });
  }

  /* ---------- registries filled by the slide code below ---------- */
  const RENDER = [];   // functions that build data-driven markup once, before icons and bindings
  const ANIM = {};     // slide id -> (slide, timeline) => signature motion
  const ACTIONS = {};  // named actions bound to keys, e.g. ACTIONS.airplane (M)
  const timers = [];
  const later = (ms, fn) => { timers.push(setTimeout(fn, ms)); };
  const every = (ms, fn, first = ms) => { timers.push(setTimeout(() => { fn(); timers.push(setInterval(fn, ms)); }, first)); };
  const stopTimers = () => { timers.forEach(id => { clearTimeout(id); clearInterval(id); }); timers.length = 0; };

  /* ---------- entrances ---------- */
  function countUp(el, tl) {
    const end = Number(el.dataset.v);
    if (!Number.isFinite(end)) return;
    const from = el.dataset.from !== undefined ? Number(el.dataset.from) : 0;
    const whole = WHOLE.has(el.dataset.f);
    const o = { v: from };
    tl.fromTo(o, { v: from }, {
      v: end, duration: Number(el.dataset.countDur || 0.9), ease: 'power2.out',
      onUpdate: () => { el.textContent = format(el, whole ? Math.round(o.v) : o.v); },
      onComplete: () => { el.textContent = format(el, end); },
    }, Number(el.dataset.countAt || 0.15));
  }

  let tl = null;
  function enter(slide) {
    if (tl) tl.kill();
    tl = null;
    stopTimers();
    if (still) return;
    tl = gsap.timeline();
    const rise = $$('[data-rise]', slide);
    if (rise.length) {
      tl.fromTo(rise, { y: 28, opacity: 0 }, { y: 0, opacity: 1, duration: 0.32, ease: 'power3.out', stagger: 0.035, clearProps: 'transform' }, 0.05);
    }
    $$('[data-count]', slide).forEach(el => countUp(el, tl));
    $$('[data-grow]', slide).forEach((el, i) => {
      tl.fromTo(el, { scaleX: 0 }, { scaleX: 1, duration: 0.6, ease: 'power3.out', transformOrigin: 'left center', clearProps: 'transform' },
        Number(el.dataset.growAt || 0.25) + i * 0.08);
    });
    if (ANIM[slide.id]) ANIM[slide.id](slide, tl);
  }

  /* ===== slide code: each later task inserts its slide's RENDER/ANIM above the end marker ===== */

  /* 01 cover: the app phones rise into place while the sonar rings pulse behind them */
  ANIM.cover = (s, t) => {
    t.fromTo($('.cover-phones', s), { y: 60, opacity: 0, scale: 0.96 }, { y: 0, opacity: 1, scale: 1, duration: 0.8, ease: 'power3.out', clearProps: 'transform' }, 0.2);
  };

  /* ===== end slide code ===== */

  /* ---------- footer, letterbox, progress ---------- */
  let lastMain = null;
  function onSlide(slide) {
    const appx = slide.hasAttribute('data-appendix');
    const bg = appx ? '#EEF4FB' : '#061233';
    document.documentElement.style.background = bg;
    document.body.style.background = bg;
    document.body.classList.toggle('on-appx', appx);
    const counter = $('#slideCounter');
    if (appx) {
      counter.textContent = 'Q&A · ' + (slide.dataset.title || '');
    } else {
      lastMain = slide;
      const main = mainSlides();
      const n = main.indexOf(slide) + 1;
      counter.textContent = `${pad(n)} / ${pad(main.length)}`;
      $('#progressFill').style.transform = `scaleX(${n / main.length})`;
    }
    enter(slide);
  }

  /* ---------- presenter timer: T shows, R restarts ---------- */
  const timerEl = $('#timer');
  let t0 = performance.now();
  function tick() {
    const s = (performance.now() - t0) / 1000;
    $('#timerLabel').textContent = Math.floor(s / 60) + ':' + pad(Math.floor(s % 60));
    $('#timerFill').style.transform = `scaleX(${Math.min(s / DECK.meta.pitchSeconds, 1)})`;
    timerEl.dataset.zone = s >= DECK.meta.timerRed ? 'red' : s >= DECK.meta.timerAmber ? 'amber' : 'ok';
  }
  setInterval(tick, 250);
  const toggleTimer = () => timerEl.classList.toggle('hidden');
  const resetTimer = () => { t0 = performance.now(); tick(); };

  /* ---------- Q&A jump menu (Q), help (H), going back (Esc / Backspace) ---------- */
  const qmenu = $('#qmenu');
  const help = $('#help');
  const isOpen = el => el.classList.contains('open');
  const setOpen = (el, open) => { el.classList.toggle('open', open); el.setAttribute('aria-hidden', String(!open)); };
  function buildQMenu() {
    const grid = $('.q-grid', qmenu);
    grid.innerHTML = '';
    sections().forEach((s, index) => {
      if (!s.hasAttribute('data-appendix')) return;
      const b = document.createElement('button');
      b.type = 'button';
      b.className = 'q-item';
      b.dataset.target = s.id;
      b.innerHTML = `<span class="q-key mono">${grid.children.length + 1}</span><span>${s.dataset.title || s.id}</span>`;
      b.addEventListener('click', () => { setOpen(qmenu, false); Reveal.slide(index); });
      grid.appendChild(b);
    });
  }
  const toggleQ = () => { setOpen(help, false); setOpen(qmenu, !isOpen(qmenu)); };
  const toggleHelp = () => { setOpen(qmenu, false); setOpen(help, !isOpen(help)); };
  function goBack() {
    if (isOpen(qmenu)) return setOpen(qmenu, false);
    if (isOpen(help)) return setOpen(help, false);
    const cur = Reveal.getCurrentSlide();
    if (cur && cur.hasAttribute('data-appendix') && lastMain) Reveal.slide(sections().indexOf(lastMain));
  }
  document.addEventListener('keydown', e => {
    if (!isOpen(qmenu)) return;
    const n = Number(e.key);
    const item = n >= 1 && n <= 9 ? $$('.q-item', qmenu)[n - 1] : null;
    if (item) { e.preventDefault(); e.stopPropagation(); item.click(); }
  }, true);

  /* ---------- ?check=1: data self-check banner for Robert ---------- */
  function runCheck() {
    if (!isCheck) return;
    const problems = DECK.verify();
    const pending = DECK.pending();
    pending.forEach(key => $$(`[data-k^="${key}"]`).forEach(el => el.classList.add('is-pending')));
    if (!problems.length && !pending.length) return;
    const list = (title, items) => items.length ? `<b>${title}</b><ul>${items.map(i => `<li>${i}</li>`).join('')}</ul>` : '';
    const banner = $('#checkBanner');
    banner.innerHTML = list('Data check failed: fix deck-data.js', problems) + list('Not yet confirmed (traction)', pending);
    banner.classList.toggle('only-pending', !problems.length);
    banner.hidden = false;
  }

  /* ---------- start ---------- */
  RENDER.forEach(fn => fn());
  inlineIcons(document);
  bind(document);
  buildQMenu();
  runCheck();

  const NEXT_KEYS = new Set([32, 34, 39, 40, 76, 78]); // space, page down, right, down, L, N
  Reveal.on('pdf-ready', () => { window.__pdfReady = true; });
  Reveal.on('slidechanged', e => onSlide(e.currentSlide));
  Reveal.initialize({
    width: 1920, height: 1080, margin: 0, minScale: 0.05, maxScale: 4,
    center: false, hash: true, controls: false, progress: false, slideNumber: false,
    overview: false, help: false, jumpToSlide: false,
    transition: 'fade', transitionSpeed: 'fast', backgroundTransition: 'none',
    pdfMaxPagesPerSlide: 1, pdfSeparateFragments: false,
    keyboardCondition: e => {
      if (isOpen(qmenu) || isOpen(help)) return [8, 27, 72, 81].includes(e.keyCode);
      const cur = Reveal.getCurrentSlide();
      return !(cur && cur.id === 'close' && NEXT_KEYS.has(e.keyCode));
    },
    keyboard: {
      8: goBack, 27: goBack, 72: toggleHelp, 81: toggleQ, 82: resetTimer, 84: toggleTimer,
      77: () => { if (ACTIONS.airplane) ACTIONS.airplane(); },
    },
    plugins: [RevealNotes],
  }).then(() => {
    onSlide(Reveal.getCurrentSlide());
    window.__deckReady = true;
  });
})();
```

```js
  /* 02 problem: 103 dots, one per fisherman. 93 fade, 10 light amber, the count falls 103 -> 10. */
  RENDER.push(function problemDots() {
    const box = $('#fisherDots');
    if (!box) return;
    const total = DECK.market.barangayFishermen;
    const lit = DECK.market.barangayGps;
    const litAt = new Set(Array.from({ length: lit }, (_, k) => Math.floor((k + 0.5) * total / lit)));
    box.innerHTML = Array.from({ length: total }, (_, i) => `<i class="dot${litAt.has(i) ? ' gps' : ''}"></i>`).join('');
    $('#gpsCount').dataset.from = String(total);
  });
  ANIM.problem = (s, t) => {
    const dots = $$('.dot', s);
    const lit = dots.filter(d => d.classList.contains('gps'));
    const dim = dots.filter(d => !d.classList.contains('gps'));
    t.fromTo(dots, { scale: 0, opacity: 0, backgroundColor: '#9FB3D9' }, { scale: 1, opacity: 1, duration: 0.3, ease: 'back.out(2)', stagger: 0.004 }, 0.25)
      .to(dim, { backgroundColor: 'rgba(159,179,217,0.22)', duration: 0.6, ease: 'power2.out', stagger: { each: 0.004, from: 'random' } }, 0.95)
      .to(lit, { backgroundColor: '#FFC857', scale: 1.25, duration: 0.4, ease: 'back.out(3)', stagger: 0.04 }, 1.2)
      .set(dots, { clearProps: 'all' });
  };

  /* 03 solution: an amber pin drops on the chart, then the tiles light in turn as the screen changes */
  ANIM.solution = (s, t) => {
    const tiles = $$('[data-tile]', s);
    const screens = $$('.scr', s);
    const phone = $('.sol-phone', s);
    let i = 0;
    const show = n => {
      tiles.forEach((el, k) => el.classList.toggle('on', k === n));
      screens.forEach((el, k) => el.classList.toggle('on', k === n));
      phone.classList.toggle('at-0', n === 0);
    };
    show(0);
    t.fromTo($('.pin-drop', s), { y: -160, opacity: 0 }, { y: 0, opacity: 1, duration: 0.6, ease: 'bounce.out', clearProps: 'transform' }, 0.35)
      .fromTo($('.pin-ripple', s), { scale: 0.2, opacity: 0.8 }, { scale: 3, opacity: 0, duration: 1, ease: 'power2.out' }, 0.9);
    every(1800, () => { i = (i + 1) % tiles.length; show(i); }, 2400);
  };
```

```js
  /* 06 traction: accounts count up, the timeline draws and each milestone pops as the line reaches it */
  RENDER.push(function tractionTimeline() {
    const ol = $('#traction .timeline');
    if (!ol) return;
    ol.innerHTML = DECK.traction.milestones.map(m =>
      `<li class="ms${m.planned ? ' planned' : ''}"><span class="ms-dot"><i data-icon="${m.icon}"></i></span><b>${m.when}</b><span>${m.what}</span></li>`).join('');
    const impl = $('#traction .impl');
    const items = DECK.traction.implemented;
    impl.hidden = !items.length;
    $('ul', impl).innerHTML = items.map(text => `<li><i data-icon="check"></i><span>${text}</span></li>`).join('');
  });
  ANIM.traction = (s, t) => {
    const dots = $$('.ms-dot', s);
    const span = 1.4;
    t.fromTo($('.tr-photo', s), { x: 60, opacity: 0 }, { x: 0, opacity: 1, duration: 0.7, ease: 'power3.out', clearProps: 'transform' }, 0.15)
      .fromTo($('.tl-fill', s), { scaleX: 0 }, { scaleX: 1, duration: span, ease: 'none', clearProps: 'transform' }, 0.5)
      .fromTo(dots, { scale: 0 }, { scale: 1, duration: 0.35, ease: 'back.out(2.5)', stagger: span / Math.max(1, dots.length - 1), clearProps: 'transform' }, 0.5);
  };

  /* 07 model: competitor five-year bars grow long, then FishPin's arrives last and tiny */
  RENDER.push(function costBars() {
    const box = $('#model .bars');
    if (!box) return;
    const max = Math.max(...DECK.competitors.map(c => c.max || c.fiveYear));
    const pct = v => `${(v / max * 100).toFixed(2)}%`;
    const php = v => 'PHP ' + nf(v);
    const rows = DECK.competitors.map(c => c.min
      ? `<div class="bar-row"><span class="who">${c.name}</span><span class="bar-track"><i class="bar-fill" data-grow style="width:${pct(c.min)}"></i><i class="bar-range" style="left:${pct(c.min)};width:${pct(c.max - c.min)}"></i><span class="bar-val" style="left:calc(${pct(c.min)} + 18px)">${php(c.min)} to ${nf(c.max)}</span></span></div>`
      : `<div class="bar-row"><span class="who">${c.name}</span><span class="bar-track"><i class="bar-fill" data-grow style="width:${pct(c.fiveYear)}"></i><span class="bar-val inside" style="left:calc(${pct(c.fiveYear)} - 18px)">${php(c.fiveYear)}</span></span></div>`);
    rows.push(`<div class="bar-row us"><span class="who">FishPin</span><span class="bar-track"><i class="bar-fill" data-grow data-grow-at="1.0" style="width:${pct(DECK.price)}"></i><span class="bar-val" style="left:calc(${pct(DECK.price)} + 18px)">${php(DECK.price)}</span></span></div>`);
    box.innerHTML = rows.join('');
    const top = DECK.competitors.filter(c => c.fiveYear).reduce((a, c) => (c.fiveYear > a.fiveYear ? c : a));
    $('#model .ratio').textContent = `${Math.floor(top.fiveYear / DECK.price)}x`;
    $('#model .ratio-who').textContent = top.name;
  });
  ANIM.model = (s, t) => {
    t.fromTo($$('.bar-row:not(.us) .bar-val', s), { opacity: 0 }, { opacity: 1, duration: 0.3, stagger: 0.08 }, 0.7)
      .fromTo($('.bar-row.us .bar-val', s), { opacity: 0 }, { opacity: 1, duration: 0.3 }, 1.6)
      .fromTo($('.cost-callout', s), { opacity: 0 }, { opacity: 1, duration: 0.4 }, 1.8);
  };
```

```js
  /* 11 impact: press-and-hold SOS. The ring fills over the hold time; release early and it drains. */
  (function sosDemo() {
    const box = $('#sos');
    if (!box) return;
    const btn = $('#sosBtn');
    const prog = $('.prog', box);
    const C = 741.4;
    let tw = null;
    const setState = s => { box.dataset.state = s; };
    function reset() {
      if (tw) tw.kill();
      tw = null;
      gsap.set(prog, { clearProps: 'strokeDashoffset,stroke' });
      setState('idle');
    }
    function start() {
      if (box.dataset.state === 'sent') reset();
      if (tw) tw.kill();
      setState('holding');
      tw = gsap.fromTo(prog, { strokeDashoffset: C }, {
        strokeDashoffset: 0, duration: DECK.product.sosHoldSeconds, ease: 'none',
        onComplete: () => { tw = null; gsap.set(prog, { clearProps: 'strokeDashoffset,stroke' }); setState('sent'); },
      });
    }
    function end() {
      if (box.dataset.state !== 'holding') return;
      if (tw) tw.kill();
      tw = gsap.to(prog, {
        strokeDashoffset: C, duration: 0.3, ease: 'power2.out',
        onComplete: () => { tw = null; gsap.set(prog, { clearProps: 'strokeDashoffset' }); setState('idle'); },
      });
    }
    btn.addEventListener('pointerdown', e => { e.preventDefault(); start(); });
    ['pointerup', 'pointerleave', 'pointercancel'].forEach(type => btn.addEventListener(type, end));
    ANIM.impact = () => { reset(); later(1000, start); };
  })();
```

```js
  RENDER.push(function appendixTables() {
```
